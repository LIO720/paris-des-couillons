// Script complet : Évaluation des pronostics, tie-breaker sur la minute du but et gestion de la cagnotte function evaluerMatch(matchReel, pronosJoueurs) { let gagnantsExacts = []; // 1. Recherche des bons scores exacts pronosJoueurs.forEach(joueur => { if ( joueur.prediction.butsDom === matchReel.butsDom && joueur.prediction.butsExt === matchReel.butsExt ) { gagnantsExacts.push(joueur); } }); // 2. Si plusieurs joueurs ont le bon score, départage par la minute du premier but (Tie-breaker) if (gagnantsExacts.length > 1) { let meilleurEcart = Infinity; let departages = []; gagnantsExacts.forEach(joueur => { const ecart = Math.abs(joueur.prediction.minutePremierBut - matchReel.minutePremierBut); if (ecart < meilleurEcart) { meilleurEcart = ecart; departages = [joueur]; } else if (ecart === meilleurEcart) { departages.push(joueur); } }); return departages; } return gagnantsExacts; } function gererCagnotte(gagnants, cagnotteActuelle) { if (gagnants && gagnants.length > 0) { return { statut: "GAGNANT", message: `Cagnotte remportée par : ${gagnants.map(g => g.nom).join(', ')} !`, montantProchainMatch: 0, gagnants: gagnants }; } else { return { statut: "PAS_DE_GAGNANT", message: "PAS DE GAGNANT POUR LES COUILLONS", montantProchainMatch: cagnotteActuelle, // Le montant est reporté au match suivant gagnants: [] }; } } // --- SIMULATION / EXEMPLE D'UTILISATION --- // Résultat réel du match const matchReel = { butsDom: 2, butsExt: 1, minutePremierBut: 24 }; // Pronostics des participants const joueurs = [ { id: 1, nom: "Thomas", prediction: { butsDom: 2, butsExt: 1, minutePremierBut: 22 } }, { id: 2, nom: "Sarah", prediction: { butsDom: 2, butsExt: 1, minutePremierBut: 40 } }, { id: 3, nom: "Marc", prediction: { butsDom: 0, butsExt: 0, minutePremierBut: 0 } } ]; let cagnotteEnCours = 150; // Montant actuel en euros // Traitement const gagnantsDuMatch = evaluerMatch(matchReel, joueurs); const resultatCagnotte = gererCagnotte(gagnantsDuMatch, cagnotteEnCours); // Affichage console (ou intégration front-end) console.log("--- RÉSULTAT DU MATCH ---"); console.log(resultatCagnotte.message); console.log(`Montant reporté pour le prochain match : ${resultatCagnotte.montantProchainMatch} €`);
